```python
"""回放验收测试：把设计方案 §09 的"需求→定位 Top-5 命中率 ≥80%"落成可执行指标。

夹具即"缩小版的真实场景"：mini（订单/商品 + PRD/checklist）+ repo-belt（库存）合并图。
失败时先怀疑检索质量，而不是改期望值——这正是验收闭环存在的目的。

import os, sys, unittest
HERE = os.path.dirname(os.path.abspath(__file__)); ROOT = os.path.dirname(HERE); sys.path.insert(0, ROOT)
from domimap import batch, scenario
from domimap.graph import load_seeds
from domimap.search import Searcher

OUT = os.path.join(HERE, "out_replay")
SEEDS = load_seeds(os.path.join(ROOT, "domimap", "seeds.json"))

def G():
    global _g
    if _g is None:
        roots = batch.discover_roots(os.path.join(HERE, "fixtures"))
        _g, _, _ = batch.build_many(roots, OUT, force=True, merged_name="fixtures")
    return _g
_g = None

# (查询文本, [Top-5 种子命中的可接受串·任一命中即算命中], 期望领域或 None)
REPLAY_REQ = [
    ("订单改价 防并发",            ["t_order", "price_version", "改价需求PRD"], "订单域"),
    ("price_version 并发控制",     ["price_version", "updatePrice"], "订单域"),
    ("order price update 需求",    ["t_order", "updatePrice", "changePrice"], "订单域"),
    ("商品 价格同步 监听",          ["PriceSyncListener", "trade-price-changed", "item-service"], "商品域"),
    ("库存预警 消费 重试",          ["stock-warn", "StockWarnConsumer", "ware-service"], "库存域"),
    ("ware_stock 表结构",          ["t_ware_stock", "sku_code", "WareStock"], None),
    ("上线 检查 清单",             ["上线checklist"], None),
    ("rocketmq 发消息 topic",       ["trade-price-changed"], None),
]
REPLAY_REFUSE = ["完全不存在的词zzzqqq", "qwer tyui asdf"]

class TestReplayAcceptance(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.g = G(); cls.s = Searcher(cls.g)

    def test_top5_hit_rate(self):
        hit = 0
        for text, accepts, _dom in REPLAY_REQ:
            r = self.s.query(text, top_k=5, expand=False)
            names = " ".join(n["name"] for n in r["seeds"])
            ok = (not r["refused"]) and any(a in names for a in accepts)
            if ok: hit += 1
            else: print("[replay MISS] %r → %s" % (text, names or "REFUSED"))
        rate = hit / len(REPLAY_REQ)
        print("[replay] Top-5 命中率 %.0f%% (%d/%d) —— 验收线 80%%" % (rate*100, hit, len(REPLAY_REQ)))
        self.assertGreaterEqual(rate, 0.8)

    def test_refusals(self):
        for text in REPLAY_REFUSE:
            self.assertTrue(self.s.query(text)["refused"], text)

    def test_domain_classify(self):
        for text, _a, dom in REPLAY_REQ:
            if dom:
                self.assertIn(dom, scenario.classify(text, SEEDS), (text, dom))

    def test_scenario_cards(self):
        card = scenario.scenario_req(self.g, self.s, "订单改价 防并发", SEEDS)
        self.assertIn("需求定位卡", card); self.assertIn("订单域", card)
        card2 = scenario.scenario_trb(self.g, self.s, "price_version 并发 覆盖")
        self.assertIn("排障卡", card2)

if __name__ == "__main__":
    unittest.main(verbosity=2)
```

```python
IDX_SCHEMA = "domimap.idx.v1"

class Searcher:
    def __init__(self, g, cache_path=None):
        self.g = g
        self.docs = {}; self.df = Counter(); self.N = 0
        if cache_path and self._load_cache(cache_path):
            return
        self._build()
        if cache_path and getattr(g, "sig", None):
            self._save_cache(cache_path)

    def _sig_ok(self, d):
        sig = getattr(self.g, "sig", None)
        return (sig and d.get("schema") == IDX_SCHEMA
                and d.get("graph_mtime") == sig["mtime"] and d.get("graph_size") == sig["size"]
                and d.get("node_count") == len(self.g.nodes))

    def _load_cache(self, path):
        try:
            with open(path, encoding="utf-8") as f:
                d = json.load(f)
        except (OSError, ValueError):
            return False
        if not self._sig_ok(d):
            return False
        self.docs = {nid: Counter(dict(pairs)) for nid, pairs in d["docs"].items()}
        self.index = defaultdict(set, {t: set(ids) for t, ids in d["index"].items()})
        self.df = Counter({t: len(v) for t, v in self.index.items()})
        self.N = d["N"]
        return True

    def _save_cache(self, path):
        sig = self.g.sig
        d = {"schema": IDX_SCHEMA, "graph_mtime": sig["mtime"], "graph_size": sig["size"],
             "node_count": len(self.g.nodes), "N": self.N,
             "docs": {nid: [[t, c] for t, c in cnt.items()] for nid, cnt in self.docs.items()},
             "index": {t: sorted(ids) for t, ids in self.index.items()}}
        tmp = path + ".tmp"
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump(d, f, ensure_ascii=False)
        os.replace(tmp, path)
```
